CREATE UNIQUE NONCLUSTERED INDEX IX_RoomSeatStudentId
ON RoomSeat (StudentId)
WHERE StudentId IS NOT NULL;

CREATE UNIQUE NONCLUSTERED INDEX IX_StudentClassRollNo
ON Student (ClassRollNo)
WHERE ClassRollNo IS NOT NULL;

CREATE UNIQUE NONCLUSTERED INDEX IX_StudentEmail
ON Student (Email)
WHERE Email IS NOT NULL;

CREATE UNIQUE NONCLUSTERED INDEX IX_StudentMobile
ON Student (Mobile)
WHERE Mobile IS NOT NULL;

CREATE UNIQUE NONCLUSTERED INDEX IX_StaffEmail
ON Staff (Email)
WHERE Email IS NOT NULL;

CREATE UNIQUE NONCLUSTERED INDEX IX_StaffMobile
ON Staff (Mobile)
WHERE Mobile IS NOT NULL;

SELECT STRING_AGG(QUOTENAME(TABLE_NAME, '"'), ', ') AS ConcatenatedTableNames
FROM INFORMATION_SCHEMA.TABLES
WHERE TABLE_TYPE = 'BASE TABLE' and TABLE_NAME <> 'sysdiagrams';
