--[dbo].[GetFormCRemittanceDetail] '1043'
CREATE PROCEDURE [dbo].[GetFormCRemittanceDetail]
	@id INT
AS
BEGIN
    SELECT  ri.*,S.Name as StatusName, Div.Name as DivisionName, Dis.Name as DistrictName, br.brName BranchName, FORMAT(ri.[CashDate], 'MMMM') AS MonthName, FORMAT(ri.[CashDate], 'dd/MM/yyyy') AS CashDatePart
	FROM [dbo].[ReceiverInfo] ri
	INNER JOIN dbo.Status S ON S.Id = ri.StatusId
	INNER JOIN dbo.District Dis ON Dis.Id = ri.DistrictId
	INNER JOIN dbo.Division Div ON Div.ID = ri.DivisionId
	INNER JOIN dbo.View_BranchInfo br ON br.brCode = ri.BranchCode
	WHERE ri.ID = @id
END

CREATE PROCEDURE [dbo].[GetRemittanceDetailPagedData]
	@BranchCode nvarchar(30),
    @page INT = NULL,
    @pageSize INT = NULL,
	@fromDate date = NULL,
	@toDate date = NULL
AS
BEGIN
    DECLARE @Offset INT = (@page - 1) * @PageSize;
	DECLARE @TotalPage INT 

	SET @TotalPage = (select COUNT(*) from [dbo].[ReceiverInfo] ri
	where 
		ri.BranchCode = @BranchCode 
		/*and ri.[CreateDate] >= CASE ISNULL(@fromDate,'') WHEN '' THEN ri.[CreateDate] ELSE @fromDate END 
		and ri.[CreateDate] <= CASE ISNULL(@toDate,'') WHEN '' THEN DATEADD(DAY, 1, ri.[CreateDate]) ELSE DATEADD(DAY, 1, @toDate) END*/

		and ri.[CreateDate] <= CASE ISNULL(@toDate,'') WHEN '' THEN DATEADD(DAY, 1, ri.[CreateDate]) ELSE DATEADD(DAY, 1, @toDate) END*/

CREATE PROCEDURE [dbo].[GetSingleOtpHistory]
@EmailOrPhone varchar(255),
@Otp varchar(20),
@timeoutInSecond int,
@dbMessage varchar(255) output
AS
BEGIN
	declare @isValid bit = 1, @TimeSpan datetime;
	
	if exists 
	(
		SELECT 1
		FROM [dbo].[OtpHistory] 
		where (Email = @EmailOrPhone or MobileNo = @EmailOrPhone) and Otp = @Otp and IsUsed <> 1 and DATEADD(SECOND, @timeoutInSecond, OtpSentTime) >= getdate()
	)
	begin
		update [dbo].[OtpHistory] set IsUsed = 1, OtpMatchedTime = GETDATE() where (Email = @EmailOrPhone or MobileNo = @EmailOrPhone) and Otp = @Otp and IsUsed <> 1
		set @dbMessage = 'SUCCESS'
	end
	else
	begin
		set @dbMessage = 'OTP has not matched, please try again.'
	end;
END
GO
